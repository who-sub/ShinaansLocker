import java.util.*;
enum numberword { 
ZERO(0, "ZERO"), ONE(1, "ONE"), TWO(2, "TWO"), THREE(3, "THREE"), FOUR(4, "FOUR"), 
FIVE(5, "FIVE"), SIX(6, "SIX"), SEVEN(7, "SEVEN"), EIGHT(8, "EIGHT"), NINE(9, "NINE"), 
TEN(10, "TEN"), ELEVEN(11, "ELEVEN"), TWELVE(12, "TWELVE"), HIRTEEN(13, 
"THIRTEEN"), FOURTEEN(14, "FOURTEEN"), FIFTEEN(15, "FIFTEEN"), SIXTEEN(16, 
"SIXTEEN"), SEVENTEEN(17, "SEVENTEEN"), EIGHTEEN(18, "EIGHTEEN"),
NINETEEN(19, "NINETEEN"), TWENTY(20, "TWENTY"), THIRTY(30, "THIRTY"), 
FORTY(40, "FORTY"), FIFTY(50, "FIFTY"), SIXTY(60, "SIXTY"), SEVENTY(70, 
"SEVENTY"), EIGHTY(80, "EIGHTY"), NINETY(90, "NINETY"); 
 
private final int value; 
private final String word; 
// Constructor 
numberword(int value, String word) { 
 this.value = value; 
 this.word = word; 
 } 
// Method to get word by value 
public static String getword(int value) { 
 for (numberword nw : numberword.values()) { 
 if (nw.value == value) { 
 return nw.word; 
 } 
 } 
 return ""; 
 } 
} 
public class numbertowords 
{ 
 public static void main(String[] args) { 
 Scanner sc = new Scanner(System.in); 
 System.out.print("Enter a number (0-99999): "); 
 int num = sc.nextInt(); 
 if (num < 0 || num > 99999) { 
 System.out.println("Out of range!"); 
 } else if (num == 0) { 
 System.out.println(numberword.ZERO.getword(0)); 
 } else { 
 System.out.println(converttowords(num).trim()); 
 } 
 }
 public static String converttowords(int n) { 
 if (n >= 1000) { 
return converttowords(n / 1000) + " THOUSAND " +" "+converttowords(n % 1000);
 } else if (n >= 100) { 
 return converttowords(n / 100) + " HUNDRED " +"AND "+ converttowords(n % 100); 
 } else if (n >= 20) { 
 return numberword.getword((n / 10) * 10) + " " + converttowords(n % 10); 
 } else if (n > 0) { 
 return numberword.getword(n); 
 } 
return ""; 
 } 
}

import java.util.*; 
public class ArrayListOperations { 
public static void main(String[] args) { 
 Scanner sc = new Scanner(System.in); 
 ArrayList list = new ArrayList<>(); 
 int choice; 
 while (true) { 
 System.out.println("\n=== ArrayList Operations Menu ==="); 
 System.out.println("1. Add elements"); 
 System.out.println("2. Sort elements"); 
 System.out.println("3. Replace an element"); 
 System.out.println("4. Remove an element"); 
 System.out.println("5. Display all elements"); 
 System.out.println("6. Add element between two elements"); 
 System.out.println("0. Exit"); 
 System.out.print("Enter choice: "); 
 
 choice = sc.nextInt(); 
 sc.nextLine(); // consume newline 
 switch (choice) { 
 case 1: // Add elements 
 System.out.print("Enter element to add: "); 
 String element = sc.nextLine(); 
 list.add(element); 
 System.out.println("Element '" + element + "' added!"); 
 break; 
 case 2: // Sort elements 
 if (!list.isEmpty()) { 
 Collections.sort(list); 
 System.out.println("Elements sorted!"); 
 } else { 
 System.out.println("List is empty!"); 
 } 
 break; 
 
case 3: // Replace element 
 if (!list.isEmpty()) { 
 System.out.print("Enter index to replace: "); 
 int index = sc.nextInt(); 
 sc.nextLine()
if (index >= 0 && index < list.size()) { 
 System.out.print("Enter new element: "); 
 String newElement = sc.nextLine(); 
 list.set(index, newElement); 
 System.out.println("Element at index " + index + " replaced!"); 
 } else { 
 System.out.println("Invalid index!"); 
 } 
 } else { 
 System.out.println("List is empty!"); 
 } 
 break; 
case 4: // Remove element 
 if (!list.isEmpty()) { 
 System.out.println("Choose removal method:"); 
 System.out.println("1. Remove by index"); 
 System.out.println("2. Remove by value"); 
 int removeChoice = sc.nextInt(); 
 sc.nextLine(); 
 if (removeChoice == 1) { 
 System.out.print("Enter index: "); 
 int index = sc.nextInt(); 
 if (index >= 0 && index < list.size()) { 
 list.remove(index); 
 System.out.println("Element removed!"); 
 } else { 
 System.out.println("Invalid index!"); 
 }
 } else if (removeChoice == 2) { 
 System.out.print("Enter element to remove: "); 
 String value = sc.nextLine(); 
 if (list.remove(value)) { 
 System.out.println("Element '" + value + "' removed!"); 
 } else { 
 System.out.println("Element not found!"); 
 } 
 } 
 } else { 
 System.out.println("List is empty!"); 
 } 
 break; 
 case 5: // Display all elements 
 System.out.println("ArrayList elements: " + list); 
 break; 
 case 6: // Add between two elements 
 if (list.size() >= 2) { 
 System.out.println("Current list: " + list); 
 System.out.print("Enter first element: "); 
 String first = sc.nextLine(); 
 System.out.print("Enter second element: "); 
 String second = sc.nextLine();
 System.out.print("Enter element to insert: "); 
 String insertElement = sc.nextLine(); 
 
 int firstIndex = list.indexOf(first); 
 int secondIndex = list.indexOf(second); 
 
 if (firstIndex != -1 && secondIndex != -1 && firstIndex < secondIndex) { 
 list.add(firstIndex + 1, insertElement); 
 System.out.println("Element inserted between '" + first + "' and '" + second + "'!"); 
 } else { 
 System.out.println("Elements not found or invalid order!"); 
 } 
 } else { 
 System.out.println("Need at least 2 elements to insert between!"); 
 } 
 break; 
 case 0: System.out.println("Exiting..."); 
 sc.close(); 
 return; 
 default: 
 System.out.println("Invalid choice!"); 
 } 
 } 
 } 
}
