def insertion_sort(arr):
 for i in range(1, len(arr)):
 key = arr[i]
 j = i - 1
 while j >= 0 and key < arr[j]:
 arr[j + 1] = arr[j]
 j -= 1

 arr[j + 1] = key

def find_min_max(arr, low, high):
 if low == high:
 return arr[low], arr[low]
 if high == low + 1:
 if arr[low] < arr[high]:
 return arr[low], arr[high]
 else:
 return arr[high], arr[low]
 mid = (low + high) // 2
 min1, max1 = find_min_max(arr, low, mid)
 min2, max2 = find_min_max(arr, mid + 1, high)
 return min(min1, min2), max(max1, max2)

import random

def merge_sort(arr):
 #if the array has 1 or 0 elements, it is already sorted
 if len(arr) <= 1:
 return arr
 # Find the middle point and divide the array into two halves
 mid = len(arr) // 2
 left_half = merge_sort(arr[:mid])
 right_half = merge_sort(arr[mid:])
 # Merge the sorted halves
 return merge(left_half, right_half)
def merge(left, right):
 sorted_array = []
 i = j = 0
 # Compare elements from both halves and add the smaller one to the result
 while i < len(left) and j < len(right):
 if left[i] < right[j]:
 sorted_array.append(left[i])
 i += 1
 else:
 sorted_array.append(right[j])
 j += 1
 # Append any remaining elements (if one half is longer than the other)
 sorted_array.extend(left[i:])
 sorted_array.extend(right[j:])
 return sorted_array

def knapsack(n, capacity, weights, prices):
 # Base condition
 if n == 0 or capacity == 0:
 return 0
 # If the weight of current item is greater than capacity,
 # we cannot include it
 if weights[n - 1] > capacity:
 return knapsack(n - 1, capacity, weights, prices)
 else:
 # Option 1: Include the current item
 include = prices[n - 1] + knapsack(
 n - 1,
 capacity - weights[n - 1],
 weights,
 prices
 )
 # Option 2: Do not include the current item
 exclude = knapsack(
 n - 1,
 capacity,
 weights,
 prices
 )
 # Choose the option with maximum profit
 return max(include, exclude)

prices = [42, 12, 40, 25]

print("Weights =", weights)
print("Prices =", prices)
print("Capacity =", capacity)
print("Optimal Solution =", knapsack(n, capacity, weights, prices))


import itertools
def totaldistance(route, distance_matrix):
 distance = 0
 for i in range(len(route) - 1):
 distance += distance_matrix[route[i]][route[i + 1]]
 distance += distance_matrix[route[-1]][route[0]]
 return distance
def tsp(distance_matrix):
 n = len(distance_matrix)
 vertices = list(range(n))
 min_distance = float('inf')
 optimal_route = []
 for permutation in itertools.permutations(vertices[1:]):
 route = [0] + list(permutation)
 current_distance = totaldistance(route, distance_matrix)
 if current_distance < min_distance:
 min_distance = current_distance
 optimal_route = route
 return optimal_route, min_distance
distance_matrix = [
 [0, 10, 15, 20],
 [10, 0, 35, 25],
 [15, 35, 0, 30],
 [20, 25, 30, 0]
]
route, distance = tsp(distance_matrix)
print("Optimal route:", route + [route[0]])
print("Minimum distance:", distance)
