function armstrong($n){
$sum = 0;
$temp = $n;
$d = strlen($n);
while($temp > 0){
$r = $temp % 10;
$sum += pow($r, $d);
$temp = (int)($temp / 10);
}return $sum == $n;}
if(isset($_POST['num'])) 
$num = $_POST['num'];
if(!is_numeric($num) || $num <= 0)

for($i=1; $i<=$num; $i++)

if ($_SERVER["REQUEST_METHOD"] == "POST") {
class DistanceCalculator {
private $feet1;
private $inches1;
private $feet2;
private $inches2;
public function construct ($feet1, $inches1, $feet2, $inches2) {
$this->feet1 = $feet1;
$this->inches1 = $inches1;
$this->feet2 = $feet2;
$this->inches2 = $inches2;
}
public function add() {
if ($this->inches1 >= 12 || $this->inches2 >= 12) {
echo "<p>Inches should be less than 12.</p>";
return;
}
$total_feet = $this->feet1 + $this->feet2;
$total_inches = $this->inches1 + $this->inches2;
if ($total_inches >= 12) {
$total_feet += floor($total_inches / 12);
$total_inches = $total_inches % 12;
}
echo "<p>Sum: " . $total_feet . " Feet\t" . $total_inches . " Inches</p>";
} 
public function difference() {
$total_inches1 = $this->feet1 * 12 + $this->inches1;
$total_inches2 = $this->feet2 * 12 + $this->inches2;
$difference = $total_inches1 - $total_inches2;
$difference_abs = abs($difference);
$difference_feet = floor($difference_abs / 12);
$difference_inches = $difference_abs % 12;
if ($difference < 0) {
echo "<p>Difference: -" . $difference_feet . " Feet\t" . $difference_inches . " Inches</p>";
} else {
echo "<p>Difference: " . $difference_feet . " Feet\t" . $difference_inches . " Inches</p>";
}
}
}
$feet1 = $_POST['feet1'];
$inches1 = $_POST['inches1'];
$feet2 = $_POST['feet2'];
$inches2 = $_POST['inches2'];
echo "<h3>Results</h3>";
$calculator = new DistanceCalculator($feet1, $inches1, $feet2, $inches2);
$calculator->add();
$calculator->difference();
}

function printMatrix($matrix) { 
echo "<table border='1'>";
foreach ($matrix as $row) {
echo "<tr>";
foreach ($row as $element) {
echo "<td>" . $element . "</td>";
}
echo "</tr>";
}
echo "</table>";

function addMatrices($matrix1, $matrix2) {
$result = [];
foreach ($matrix1 as $i => $row) {
foreach ($row as $j => $element) {
$result[$i][$j] = $element + $matrix2[$i][$j];
}
}
return $result;
}
function multiplyMatrices($matrix1, $matrix2) {
$result = [];
$rows1 = count($matrix1);
$cols1 = count($matrix1[0]);
$cols2 = count($matrix2[0]);
for ($i = 0; $i < $rows1; $i++) {
for ($j = 0; $j < $cols2; $j++) {
$result[$i][$j] = 0;
for ($k = 0; $k < $cols1; $k++) {
$result[$i][$j] += $matrix1[$i][$k] * $matrix2[$k][$j];
}
} 
return $result;
